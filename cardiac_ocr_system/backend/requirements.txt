# --- Extraction engines -----------------------------------------------------
# PRIMARY: Groq vision (app/ocr/groq_extractor.py) reads the report image and returns
# label/value pairs. It needs NO extra package -- it calls the Groq REST API via httpx (above),
# deliberately avoiding an SDK so no new dependency can re-introduce a version conflict.
# Configure GROQ_API_KEY / GROQ_VISION_MODEL in backend/.env.
#
# SECOND READER: PaddleOCR independently re-reads the same image so every Groq value can be
# corroborated (app/ocr/cross_validation.py). This is what supplies REAL confidence scores --
# a vision LLM returns none, and fabricating them is not acceptable for clinical data.
paddleocr==2.7.3
paddlepaddle==2.6.2

# !! DO NOT UNPIN protobuf !!
# paddlepaddle 2.6.2 requires protobuf<=3.20.2 on Windows. Left unpinned, a later
# `pip install -r requirements.txt` resolves protobuf upward and breaks PaddleOCR with
# "Descriptors cannot be created directly" at import time.
